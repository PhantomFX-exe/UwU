        local function getCharacterBounds(character)
            local minX,minY,minZ = math.huge,math.huge,math.huge
            local maxX,maxY,maxZ = -math.huge,-math.huge,-math.huge
            local found = false
            for _,part in ipairs(character:GetChildren()) do
                if part:IsA('BasePart') and part.Name ~= 'Head' then
                    found = true
                    local p,s = part.Position, part.Size
                    minX=math.min(minX,p.X-s.X/2) minY=math.min(minY,p.Y-s.Y/2) minZ=math.min(minZ,p.Z-s.Z/2)
                    maxX=math.max(maxX,p.X+s.X/2) maxY=math.max(maxY,p.Y+s.Y/2) maxZ=math.max(maxZ,p.Z+s.Z/2)
                end
            end
            local head = character:FindFirstChild('Head')
            if head then
                found = true
                local p = head.Position
                local hs = 0.5
                minX=math.min(minX,p.X-hs) minY=math.min(minY,p.Y-hs) minZ=math.min(minZ,p.Z-hs)
                maxX=math.max(maxX,p.X+hs) maxY=math.max(maxY,p.Y+hs) maxZ=math.max(maxZ,p.Z+hs)
            end
            if not found then return nil end
            return Vector3.new(minX,minY,minZ), Vector3.new(maxX,maxY,maxZ)
        end

            local botCenter = Vector3.new((minB.X+maxB.X)/2,minB.Y,(minB.Z+maxB.Z)/2)

            local sBot,onBot      = w2s(botCenter)

            local width  = height*0.6

    local function GetClosestPlayer()
        local closestPlayer = nil
        local shortestDistance = AimbotSettings.FOV
        local screenCenter = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local targetPart = plr.Character:FindFirstChild(AimbotSettings.TargetPart)
                local hum = plr.Character:FindFirstChild("Humanoid")
                if targetPart and hum and hum.Health > 0 then
                    local pos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                    if onScreen then
                        local dist = (Vector2.new(pos.X, pos.Y) - screenCenter).Magnitude
                        if dist < shortestDistance then closestPlayer = plr shortestDistance = dist end
                    end
                end
            end
        end
        return closestPlayer
    end

                local spd = CharacterSettings.FlySpeed
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then md = md + Camera.CFrame.LookVector * spd end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then md = md - Camera.CFrame.LookVector * spd end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then md = md - Camera.CFrame.RightVector * spd end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then md = md + Camera.CFrame.RightVector * spd end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then md = md + Vector3.new(0,spd,0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then md = md - Vector3.new(0,spd,0) end
                hrp.Velocity = md
            end
        end
        local shouldAim = AimbotSettings.Enabled and Options.AimbotKey and Options.AimbotKey:GetState()
        if shouldAim then
            local target
            if AimbotSettings.Sticky and AimbotSettings.StickyTarget then
                local sc = AimbotSettings.StickyTarget
                if sc.Character and sc.Character:FindFirstChild("Humanoid") and sc.Character.Humanoid.Health > 0 then
                    target = sc
                else
                    AimbotSettings.StickyTarget = nil
                end
            end
            if not target then
                target = GetClosestPlayer()
                if AimbotSettings.Sticky and target then AimbotSettings.StickyTarget = target end
            end
            if target and target.Character then
                local tp = target.Character:FindFirstChild(AimbotSettings.TargetPart)
                if tp then
                    Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, tp.Position), AimbotSettings.Smoothness)
                end
            end
        else
            AimbotSettings.StickyTarget = nil
        end
        if MiscSettings.AntiStompEnabled and LocalPlayer.Character then
            local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
            local be  = LocalPlayer.Character:FindFirstChild("BodyEffects")
            local ko  = be and be:FindFirstChild("K.O")
            if hum and ko and ko.Value then
                if respawnEnabled then
                    local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then lastDeathPos = hrp.CFrame end
                end
                hum.Health = 0
            end
            if hum and hum.Health <= 1 then
                if respawnEnabled then
                    local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then lastDeathPos = hrp.CFrame end
                end
                hum.Health = 0
            end
        end
        if MiscSettings.NoClipEnabled and LocalPlayer.Character then
            for _, part in pairs(LocalPlayer.Character:GetChildren()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end)

    local function scanShop()
        local ignored = workspace:FindFirstChild("Ignored")
        if not ignored then return {} end
        local shop = ignored:FindFirstChild("Shop")
        if not shop then return {} end
        for _, item in ipairs(shop:GetChildren()) do
            local name = string.match(item.Name, "%b[]")
            local head = item:FindFirstChild("Head")
            if head and head.CFrame.p.Y > -35 and name then
                name = name:sub(2,-2)
                if name:find("Ammo") then
                    local base = name:sub(1,-6)
                    if not all_items[base] then all_items[base]={main=nil,ammo=head}
                    elseif not all_items[base].ammo then all_items[base].ammo=head end
                else
                    if not all_items[name] then all_items[name]={main=head,ammo=nil}
                    elseif not all_items[name].main then all_items[name].main=head end
                end
            end
        end

    local function _canBuyArmor()
        local character = _armorPlayer.Character

        local bodyEffects = character:FindFirstChild("BodyEffects")

    local function _teleportAndBuy(shopItem)
        local character = _armorPlayer.Character
        if not character then getgenv().autoArmorBuying = false return end
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then getgenv().autoArmorBuying = false return end
        local targetPart = shopItem:FindFirstChild("Head")
        if not targetPart then
            for _, v in pairs(shopItem:GetDescendants()) do
                if v:IsA("BasePart") then targetPart = v break end
            end
        end
        local clickDetector = shopItem:FindFirstChildOfClass("ClickDetector")
        if not clickDetector then
            for _, v in pairs(shopItem:GetDescendants()) do
                if v:IsA("ClickDetector") then clickDetector = v break end
            end
        end
        if not targetPart or not clickDetector then getgenv().autoArmorBuying = false return end
        local originalCFrame = hrp.CFrame
        hrp.CFrame = targetPart.CFrame * CFrame.new(0,3.1,0) task.wait(0.15)
        fireclickdetector(clickDetector) task.wait(0.15)
        hrp.CFrame = originalCFrame task.wait(0.5)
        getgenv().autoArmorBuying = false
    end

        if not character then _lastHP = nil return end
        local humanoid = character:FindFirstChild("Humanoid")
        if not humanoid then return end
        local currentHP = humanoid.Health
        if _lastHP == nil then _lastHP = currentHP return end
        if currentHP < _lastHP then
            _lastHP = currentHP

                if shopItem then
                    getgenv().autoArmorBuying = true
                    task.spawn(function() _teleportAndBuy(shopItem) end)
                end
            end
            return
        end
        _lastHP = currentHP
    end)
